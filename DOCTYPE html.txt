<body>

    <div id="game-container">
        <div id="score">Score: 0</div>
        <div id="player"></div>
        <div id="obstacle"></div>
        <div id="game-over">GAME OVER<br><span style="font-size: 16px;">Press Space to Restart</span></div>
    </div>

    <script>
        const player = document.getElementById('player');
        const obstacle = document.getElementById('obstacle');
        const scoreDisplay = document.getElementById('score');
        const gameOverText = document.getElementById('game-over');

        let isJumping = false;
        let gravity = 0.9;
        let score = 0;
        let isGameOver = false;
        let obstacleSpeed = 6;
        let obstaclePosition = 800;

        // Jump Controls
        document.addEventListener('keydown', function(event) {
            if (event.code === 'Space' || event.code === 'ArrowUp') {
                if (isGameOver) {
                    restartGame();
                } else if (!isJumping) {
                    jump();
                }
            }
        });

        function jump() {
            isJumping = true;
            let position = 0;
            
            // Upward movement & gravity loop
            let timerId = setInterval(function() {
                // Reach peak
                if (position >= 130) {
                    clearInterval(timerId);

                    // Falling down
                    let downTimerId = setInterval(function() {
                        if (position <= 0) {
                            clearInterval(downTimerId);
                            isJumping = false;
                        }
                        position -= 5;
                        player.style.bottom = position + 'px';
                    }, 20);
                }

                // Going up
                position += 8;
                player.style.bottom = position + 'px';
            }, 20);
        }

        function gameLoop() {
            if (isGameOver) return;

            // Move obstacle
            obstaclePosition -= obstacleSpeed;
            if (obstaclePosition < -30) {
                obstaclePosition = 800; // Reset obstacle to right side
                score += 10;
                scoreDisplay.innerText = 'Score: ' + score;

                // Slightly speed up game as score increases
                if (obstacleSpeed < 15) obstacleSpeed += 0.2; 
            }
            obstacle.style.left = obstaclePosition + 'px';

            // Hit detection
            let playerBottom = parseInt(window.getComputedStyle(player).getPropertyValue('bottom'));
            
            if (obstaclePosition > 50 && obstaclePosition < 90 && playerBottom < 50) {
                gameOver();
            }

            requestAnimationFrame(gameLoop);
        }

        function gameOver() {
            isGameOver = true;
            gameOverText.style.display = 'block';
        }

        function restartGame() {
            isGameOver = false;
            score = 0;
            obstacleSpeed = 6;
            obstaclePosition = 800;
            scoreDisplay.innerText = 'Score: 0';
            gameOverText.style.display = 'none';
            requestAnimationFrame(gameLoop);
        }

        // Start the game
        requestAnimationFrame(gameLoop);
    </script>
